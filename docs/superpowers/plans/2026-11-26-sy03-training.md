# SY-03 · Entrenamiento persistido — plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** La rutina, sus ediciones, las sesiones terminadas, el historial, las decisiones de carga y los temporizadores por ejercicio se guardan en Supabase y sobreviven a reinstalar la app.

**Architecture:** Igual que SY-02: repositorio + mappers puros + hooks de TanStack Query. `useProfile()` sobrepone `trainingPlan` y `nextLoads` del servidor y `useProgressStore()` sobrepone `workouts`, así las pantallas que leen esos campos no cambian. Las escrituras de varias tablas van por RPC (`save_workout_plan` nueva, `submit_session` existente). La sesión terminada se guarda primero en el teléfono como "pendiente" y se borra solo cuando el servidor devuelve su id; si falla, Inicio ofrece reintentar.

**Tech Stack:** Expo Router, TanStack Query 5, Zustand, supabase-js 2.118, zod 4, Vitest, pgTAP (Supabase CLI).

**Spec:** `docs/superpowers/specs/2026-11-24-account-persistence-design.md` (sección SY-03 y decisiones 2 a 4).

```sql
-- Plan de entrenamiento atómico (RF-05, RF-19, RNF-08): archiva el plan activo e inserta el
-- nuevo con sus días, ejercicios y la traza de la recomendación, en una transacción.
-- security invoker: corre con la RLS de quien llama (exige consentimiento vigente).
create function public.save_workout_plan(payload jsonb) returns uuid
language plpgsql security invoker set search_path = '' as $$
declare
  new_plan_id uuid;
  new_day_id uuid;
  day jsonb;
begin
  update public.workout_plans set status = 'archived'
  where user_id = (select auth.uid()) and status = 'active';

  insert into public.workout_plans (rule_version, template_id, template_version, catalog_version, environment, level,
    days_per_week, session_minutes, equipment, zones, changes, warnings)
  values (
    payload ->> 'rule_version', payload ->> 'template_id', payload ->> 'template_version', payload ->> 'catalog_version',
    payload ->> 'environment', payload ->> 'level', (payload ->> 'days_per_week')::smallint, (payload ->> 'session_minutes')::smallint,
    array(select jsonb_array_elements_text(payload -> 'equipment')),
    array(select jsonb_array_elements_text(coalesce(payload -> 'zones', '[]'))),
    coalesce(payload -> 'changes', '[]'),
    array(select jsonb_array_elements_text(coalesce(payload -> 'warnings', '[]')))
  )
  returning id into new_plan_id;

  for day in select value from jsonb_array_elements(payload -> 'days') loop
    insert into public.workout_days (plan_id, position, weekday, name)
    values (new_plan_id, (day ->> 'position')::smallint, (day ->> 'weekday')::smallint, day ->> 'name')
    returning id into new_day_id;

    insert into public.workout_exercises (day_id, position, exercise_id, sets, reps_min, reps_max, rest_seconds, tempo, note)
    select new_day_id, e.position, e.exercise_id, e.sets, e.reps_min, e.reps_max, e.rest_seconds,
      case when e.tempo is null or jsonb_typeof(e.tempo) <> 'array' then null
        else array(select t::smallint from jsonb_array_elements_text(e.tempo) as t) end,
      e.note
    from jsonb_to_recordset(coalesce(day -> 'exercises', '[]')) as e (
      position smallint, exercise_id text, sets smallint, reps_min smallint, reps_max smallint,
      rest_seconds smallint, tempo jsonb, note text
    );
  end loop;

  insert into public.recommendation_runs (kind, rule_version, params, result_id)
  values ('workout-plan', payload ->> 'rule_version', payload - 'days', new_plan_id);

  return new_plan_id;
end $$;

revoke execute on function public.save_workout_plan(jsonb) from public, anon;
grant execute on function public.save_workout_plan(jsonb) to authenticated;
```

```ts
// UUID v4 para claves idempotentes creadas en el teléfono (client_id de submit_session).
// No es un secreto: solo evita duplicar un envío. Hermes no siempre trae crypto.getRandomValues.
export function newClientId(): string {
  const bytes = new Uint8Array(16);
  const crypto = (globalThis as { crypto?: { getRandomValues?: (array: Uint8Array) => Uint8Array } }).crypto;
  if (crypto?.getRandomValues) crypto.getRandomValues(bytes);
  else for (let index = 0; index < bytes.length; index += 1) bytes[index] = Math.floor(Math.random() * 256);
  bytes[6] = (bytes[6]! & 0x0f) | 0x40;
  bytes[8] = (bytes[8]! & 0x3f) | 0x80;
  const hex = Array.from(bytes, (byte) => byte.toString(16).padStart(2, '0')).join('');
  return `${hex.slice(0, 8)}-${hex.slice(8, 12)}-${hex.slice(12, 16)}-${hex.slice(16, 20)}-${hex.slice(20)}`;
}
```

```ts
// Series completadas de ejercicios del catálogo. En los de tiempo, reps son segundos.
export function sessionPayload(state: SessionState, meta: { clientId: string; planId: string | null; dayId: string | null; finishedAt: Date }) {
  return {
    client_id: meta.clientId, plan_id: meta.planId, day_id: meta.dayId, title: state.title,
    started_at: new Date(meta.finishedAt.getTime() - state.elapsedMs).toISOString(), finished_at: meta.finishedAt.toISOString(),
    sets: state.exercises.flatMap((exercise, position) => {
      const exerciseId = exercise.exerciseId;
      if (!exerciseId) return [];
      const duration = exercise.measure === 'duration';
      return exercise.sets.filter((set) => set.completed).map((set) => ({
        exercise_id: exerciseId, exercise_position: position, set_number: set.id,
        weight_kg: duration ? null : set.weightKg, reps: duration ? null : set.reps, duration_seconds: duration ? set.reps : null,
        completed: true, kind: set.calibration ? 'calibration' as const : 'work' as const,
        reserve: set.reserve ?? null, estimated: set.estimated ?? false,
      }));
    }),
  };
}
export type SessionPayload = ReturnType<typeof sessionPayload>;

export type SessionWithSets = SessionRow & { performed_sets: SetRow[] };

export function workoutsFromRows(sessions: SessionWithSets[]): WorkoutRecord[] {
  return sessions.map((session) => {
    const finished = new Date(session.finished_at);
    const sets = session.performed_sets.filter((set) => set.completed)
      .sort((a, b) => a.exercise_position - b.exercise_position || a.set_number - b.set_number);
    const exercises: NonNullable<WorkoutRecord['exercises']> = [];
    for (const set of sets) {
      const entry = { weightKg: set.weight_kg ?? 0, reps: set.reps ?? set.duration_seconds ?? 0 };
      const last = exercises.at(-1);
      if (last?.exerciseId === set.exercise_id) last.sets.push(entry);
      else exercises.push({ exerciseId: set.exercise_id, sets: [entry] });
    }
    return {
      id: session.client_id, date: todayDateKey(finished), time: currentTimeLabel(finished), name: session.title,
      durationMinutes: Math.max(1, Math.round((finished.getTime() - new Date(session.started_at).getTime()) / 60_000)),
      volumeKg: sets.reduce((sum, set) => sum + (set.weight_kg ?? 0) * (set.reps ?? 0), 0), status: 'completed', exercises,
    };
  });
}

// BR-22: manda la decisión más reciente de cada ejercicio. Una carga aplicada o modificada
// queda pendiente hasta que una sesión posterior incluye ese ejercicio.
export function nextLoadsFromRows(suggestions: SuggestionRow[], sessions: SessionWithSets[]): Record<string, number> {
  const lastSession = new Map<string, string>();
  for (const session of sessions) {
    for (const set of session.performed_sets) {
      const seen = lastSession.get(set.exercise_id);
      if (!seen || session.finished_at > seen) lastSession.set(set.exercise_id, session.finished_at);
    }
  }
  const latest = new Map<string, SuggestionRow>();
  for (const row of suggestions) {
    const seen = latest.get(row.exercise_id);
    if (!seen || row.created_at > seen.created_at) latest.set(row.exercise_id, row);
  }
  const loads: Record<string, number> = {};
  for (const [exerciseId, row] of latest) {
    const accepted = (row.decision === 'applied' || row.decision === 'modified') && row.applied_kg !== null;
    const after = lastSession.get(exerciseId);
    if (accepted && (!after || new Date(row.created_at) > new Date(after))) loads[exerciseId] = row.applied_kg!;
  }
  return loads;
}

  decision: decision.decision, rule_version: ruleVersion, decided_at: now.toISOString(),
});

  ? { exercise_id: exerciseId, kind: 'interval' as const, rest_seconds: restSeconds, work_seconds: spec.workSeconds,
    interval_rest_seconds: spec.restSeconds, rounds: spec.rounds, hold_seconds: null }
  : { exercise_id: exerciseId, kind: 'hold' as const, rest_seconds: restSeconds, hold_seconds: spec.seconds,
    work_seconds: null, interval_rest_seconds: null, rounds: null };

export function timerSpecFromRow(row: TimerRow): TimerSpec | null {
  if (row.kind !== 'interval' || row.work_seconds === null || row.interval_rest_seconds === null || row.rounds === null) return null;
  return { kind: 'interval', workSeconds: row.work_seconds, restSeconds: row.interval_rest_seconds, rounds: row.rounds };
}
```

```ts
import type { Json } from '@strongerfit/contracts/database';
import type { Plan, PlannedExercise } from '@strongerfit/training-engine';
import { db, requireUserId } from '../../../lib/data/client';
import { unwrap } from '../../../lib/data/errors';
import type { WorkoutRecord } from '../../progress/domain/progress';

import type { TimerSpec } from '../domain/timers';
import { exerciseColumns, loadDecisionRow, nextLoadsFromRows, planFromRows, planPayload, timerPresetRow, timerSpecFromRow,
  workoutsFromRows, type SessionPayload, type SessionWithSets, type StoredPlan } from './training.mappers';

// Historial reciente: suficiente para récords, cargas y gráficos de un año de uso típico.
const HISTORY_LIMIT = 300;

export async function fetchActivePlan(): Promise<StoredPlan | null> {
  const plan = unwrap(await db().from('workout_plans').select('*').eq('status', 'active').maybeSingle());
  if (!plan) return null;
  const days = unwrap(await db().from('workout_days').select('*').eq('plan_id', plan.id));
  const exercises = days.length
    ? unwrap(await db().from('workout_exercises').select('*').in('day_id', days.map((day) => day.id)))
    : [];
  return planFromRows(plan, days, exercises);
}

export async function saveWorkoutPlan(plan: Plan): Promise<string> {
  return unwrap(await db().rpc('save_workout_plan', { payload: planPayload(plan) as unknown as Json }));
}

export async function updatePlanExercise(dayId: string, index: number, exercise: PlannedExercise): Promise<void> {
  const { position: _position, ...columns } = exerciseColumns(exercise, index);
  unwrap(await db().from('workout_exercises').update(columns).eq('day_id', dayId).eq('position', index));
}

export async function submitSession(payload: SessionPayload): Promise<string> {
  return unwrap(await db().rpc('submit_session', { payload: payload as unknown as Json }));
}

export async function fetchTrainingHistory(): Promise<{ workouts: WorkoutRecord[]; nextLoads: Record<string, number> }> {
  const [sessions, suggestions] = await Promise.all([
    db().from('workout_sessions').select('*, performed_sets(*)').order('finished_at', { ascending: false }).limit(HISTORY_LIMIT),
    db().from('load_suggestions').select('*').order('created_at', { ascending: false }).limit(HISTORY_LIMIT),
  ]);
  const rows = unwrap(sessions) as SessionWithSets[];
  return { workouts: workoutsFromRows(rows), nextLoads: nextLoadsFromRows(unwrap(suggestions), rows) };
}

export async function recordLoadDecision(decision: OverloadDecision, ruleVersion: string): Promise<void> {
  const row = unwrap(await db().from('load_suggestions').insert(loadDecisionRow(decision, ruleVersion, new Date())).select('id').single());
  unwrap(await db().from('recommendation_runs').insert({
    kind: 'load-suggestion', rule_version: ruleVersion, result_id: row.id,
    params: { exercise_id: decision.exerciseId, suggested_kg: decision.suggestedKg },
  }));
}

export async function fetchTimerPresets(): Promise<Record<string, TimerSpec>> {
  const rows = unwrap(await db().from('timer_presets').select('*'));
  return Object.fromEntries(rows.flatMap((row) => {
    const spec = timerSpecFromRow(row);
    return spec ? [[row.exercise_id, spec]] : [];
  }));
}

export async function saveTimerPreset(exerciseId: string, restSeconds: number, spec: TimerSpec): Promise<void> {
  await requireUserId();
  unwrap(await db().from('timer_presets').upsert(timerPresetRow(exerciseId, restSeconds, spec), { onConflict: 'user_id,exercise_id' }));
}
```

```ts
import { useQueryClient } from '@tanstack/react-query';
import { useFocusEffect } from 'expo-router';
import { useCallback, useContext, useRef, useState } from 'react';
import { dataErrorMessage, toDataError } from '../../lib/data/errors';
import { clearPendingSession, readPendingSession, writePendingSession } from '../../storage/pendingSession';
import { SessionContext } from '../account/SessionContext';
import { submitSession } from './data/training.repository';
import { parsePending, type PendingSession } from './domain/pending';
import { trainingKey } from './useTraining';

// Envía la sesión terminada. Primero queda guardada en el teléfono; se borra al confirmar el
// servidor. Si falla, queda pendiente y Inicio ofrece reintentar (RNF-06, CA-03).
export function usePendingSession() {
  const { userId } = useContext(SessionContext);
  const client = useQueryClient();
  const [pending, setPending] = useState<PendingSession | null>(null);
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const inFlight = useRef(false);

  const send = useCallback(async (session: PendingSession) => {
    if (inFlight.current) return;
    inFlight.current = true;
    setSubmitting(true);
    setError(null);
    try {
      await submitSession(session.payload);
      await clearPendingSession().catch(() => undefined);
      setPending(null);
      await client.invalidateQueries({ queryKey: trainingKey(userId ?? 'anon') });
    } catch (failure) {
      setPending(session);
      setError(dataErrorMessage(toDataError(failure)));
    } finally {
      inFlight.current = false;
      setSubmitting(false);
    }
  }, [client, userId]);

  const submit = useCallback((session: PendingSession) => {
    setPending(session);
    writePendingSession(JSON.stringify(session)).catch(() => undefined).finally(() => { void send(session); });
  }, [send]);

  // Al volver a la pantalla, si quedó una sesión sin enviar se intenta una vez.
  useFocusEffect(useCallback(() => {
    let active = true;
    readPendingSession().then(parsePending).then((saved) => {
      if (!active || !saved) return;
      setPending(saved);
      void send(saved);
    }).catch(() => undefined);
    return () => { active = false; };
  }, [send]));

  const retry = useCallback(() => { if (pending) void send(pending); }, [pending, send]);
  return { pending, submit, retry, submitting, error };
}
```
